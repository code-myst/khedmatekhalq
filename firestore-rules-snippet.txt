// তাই Console-এ আপনার বর্তমান rules মুছে এটা পেস্ট করবেন না।
// বর্তমান rules-এর `match /databases/{database}/documents { ... }` ব্লকের ভেতরে নিচের অংশটুকু যোগ করুন
// (পুরোনো কোনো নিয়ম মুছবেন না), তারপর Publish করুন।

    // ---------- খেদমতে খালক বইয়ের দোকান (kb_*) ----------
    function kbAdmin() {
      return request.auth != null
        && exists(/databases/$(database)/documents/kb_admins/$(request.auth.uid));
    }

    // অ্যাডমিন তালিকা: ব্যবহারকারী শুধু নিজের ডকুমেন্ট পড়তে পারে; লেখা যাবে শুধু Console থেকে
    match /kb_admins/{uid} {
      allow read: if request.auth != null && request.auth.uid == uid;
      allow write: if false;
    }

    match /kb_products/{id} {
      allow read: if true;
      allow write: if kbAdmin();
    }

    match /kb_categories/{id} {
      allow read: if true;
      allow write: if kbAdmin();
    }

    // রিভিউ: ডকুমেন্ট আইডি = productId_uid (একজন একটিই রিভিউ দিতে পারে)
    match /kb_reviews/{rid} {
      allow read: if true;
      allow create, update: if request.auth != null
        && request.resource.data.userId == request.auth.uid
        && rid == request.resource.data.productId + '_' + request.auth.uid
        && request.resource.data.rating is number
        && request.resource.data.rating >= 1
        && request.resource.data.rating <= 5
        && request.resource.data.text is string
        && request.resource.data.text.size() <= 1000;
      allow delete: if kbAdmin()
        || (request.auth != null && resource.data.userId == request.auth.uid);
    }

    // অর্ডার: যে কেউ (গেস্টসহ) নতুন অর্ডার দিতে পারবে, কিন্তু পড়তে পারবে শুধু মালিক ও অ্যাডমিন
    match /kb_orders/{oid} {
      allow create: if request.resource.data.status == 'pending'
        && request.resource.data.items is list
        && request.resource.data.items.size() > 0
        && request.resource.data.items.size() <= 50
        && request.resource.data.total is number
        && request.resource.data.total >= 0
        && request.resource.data.name is string
        && request.resource.data.name.size() <= 120
        && request.resource.data.phone is string
        && request.resource.data.address is string
        && (request.resource.data.userId == null
            || (request.auth != null && request.resource.data.userId == request.auth.uid));
      allow read: if (request.auth != null && resource.data.userId == request.auth.uid) || kbAdmin();
      allow update, delete: if kbAdmin();
    }
